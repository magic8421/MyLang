不要这样 ObjHeader 加一个 type_id 字段 32位无符号 用其最高位 做一个 TYPE_IS_ARRAY bit    
type_id 低24位可以用来存 type信息 
目前就预留 16个类型给 简单数据 比如
`u8`, `u16`, `u32`, `u64`,  `i8`, `i16`, `i32`, `i64`, `f32`, `f64`
所以class的type_id要大于15  编译时 按照定义出现的顺序 每个class type_id递增  不重复 
mylang_array_release 释放的时候 判断 type_id是否大于15  就可以决定要不要 mylang_release(data[i])

inline int array_get_int(int *arr, size_t idx) {
    if ((size_t)(idx) >= *(size_t*)((char*)(arr) - sizeof(size_t))){
        __debugbreak();
    }
    return arr[idx];
}

inline void array_set_int(int *arr, size_t idx, int val) {
    if ((size_t)(idx) >= *(size_t*)((char*)(arr) - sizeof(size_t))) { 
        __debugbreak();
    }
    arr[idx] = val;
}

第二种方法：

typedef struct {
    size_t len;
    int data[];
} IntArray;

inline int array_get_int(IntArray *arr, size_t idx) {
    if (idx >= arr->len) {
        __debugbreak();
    }
    return arr->data[idx];
}

inline void array_set_int(IntArray *arr, size_t idx, int val) {
    if (idx >= arr->len) {
        __debugbreak();
    }
    arr->data[idx] = val;
}
